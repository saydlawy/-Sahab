import com.saydlawy.sahab.data.catalog.CatalogDatabase
import com.saydlawy.sahab.data.channel.ChannelDatabase
import org.junit.Assert.assertEquals
import org.junit.Assert.assertFalse
import org.junit.Assert.assertTrue
import org.junit.Test
import java.util.Locale

class DatabaseMigrationRegressionTest {

    /**
     * Lightweight SQLite schema + data simulator implementing [SupportSQLiteDatabase] so Room
     * migrations (`ChannelDatabase.MIGRATION_1_2` and `CatalogDatabase.MIGRATION_1_2..3_4`) can
     * be executed and verified in pure JVM unit tests without requiring an emulator.
     */
    private class RecordingSqliteDatabase : SupportSQLiteDatabase {
        val tables = linkedMapOf<String, MutableSet<String>>()
        val indices = linkedMapOf<String, Boolean>() // indexName -> unique
        val tableRows = linkedMapOf<String, MutableList<Map<String, Any?>>>()
        val executedSql = mutableListOf<String>()

        fun defineTable(name: String, columns: List<String>, rows: List<Map<String, Any?>> = emptyList()) {
            tables[name] = columns.toMutableSet()
            tableRows[name] = rows.map { it.toMap() }.toMutableList()
        }

        override fun execSQL(sql: String) {
            val normalized = sql.trim().replace(Regex("\\s+"), " ")
            executedSql += normalized

            val addColRegex = Regex(
                "ALTER TABLE `?(\\w+)`? ADD COLUMN `?(\\w+)`?.*",
                RegexOption.IGNORE_CASE,
            )
            val renameTableRegex = Regex(
                "ALTER TABLE `?(\\w+)`? RENAME TO `?(\\w+)`?",
                RegexOption.IGNORE_CASE,
            )
            val createTableRegex = Regex(
                "CREATE TABLE IF NOT EXISTS `?(\\w+)`? \\((.+)\\)",
                RegexOption.IGNORE_CASE,
            )
            val dropTableRegex = Regex(
                "DROP TABLE (?:IF EXISTS )?`?(\\w+)`?",
                RegexOption.IGNORE_CASE,
            )
            val dropIndexRegex = Regex(
                "DROP INDEX IF EXISTS `?(\\w+)`?",
                RegexOption.IGNORE_CASE,
            )
            val createIndexRegex = Regex(
                "CREATE (UNIQUE )?INDEX IF NOT EXISTS `?(\\w+)`? ON `?(\\w+)`? \\((.+)\\)",
                RegexOption.IGNORE_CASE,
            )
            val migrateLegacyListsRegex = Regex(
                "INSERT OR IGNORE INTO `channel_lists` .* FROM `channel_lists_legacy`",
                RegexOption.IGNORE_CASE,
            )

            addColRegex.matchEntire(normalized)?.let { m ->
                val table = m.groupValues[1]
                val col = m.groupValues[2]
                tables.getOrPut(table) { linkedSetOf() }.add(col)
                return
            }

            renameTableRegex.matchEntire(normalized)?.let { m ->
                val oldName = m.groupValues[1]
                val newName = m.groupValues[2]
                tables.remove(oldName)?.let { tables[newName] = it }
                tableRows.remove(oldName)?.let { tableRows[newName] = it }
                return
            }

            createTableRegex.matchEntire(normalized)?.let { m ->
                val table = m.groupValues[1]
                val body = m.groupValues[2]
                if (!tables.containsKey(table)) {
                    val cols = body.split(",")
                        .map { it.trim() }
                        .filterNot { it.uppercase(Locale.US).startsWith("PRIMARY KEY") }
                        .mapNotNull { def ->
                            Regex("^`?(\\w+)`?\\s+").find(def)?.groupValues?.get(1)
                        }
                        .toMutableSet()
                    tables[table] = cols
                    tableRows.putIfAbsent(table, mutableListOf())
                }
                return
            }

            if (migrateLegacyListsRegex.containsMatchIn(normalized)) {
                val legacyRows = tableRows["channel_lists_legacy"].orEmpty()
                val targetRows = tableRows.getOrPut("channel_lists") { mutableListOf() }
                for (row in legacyRows) {
                    targetRows += mapOf(
                        "listId" to row["id"],
                        "name" to row["name"],
                        "icon" to "",
                        "sortOrder" to row["sortOrder"],
                        "sortMode" to "custom",
                        "createdAt" to row["updatedAt"],
                        "updatedAt" to row["updatedAt"],
                    )
                }
                return
            }

            dropTableRegex.matchEntire(normalized)?.let { m ->
                val table = m.groupValues[1]
                tables.remove(table)
                tableRows.remove(table)
                return
            }

        val pin = "4829"
        val uid = "user_uid_123"

        val cipher1 = CloudAccountCrypto.encryptV2(secretPayload, pin)
        val cipher2 = CloudAccountCrypto.encryptV2(secretPayload, pin)

        assertTrue(cipher1.startsWith("enc_v2:"))
        // Random 16-byte salt & 12-byte IV guarantee distinct ciphertexts for identical plaintexts
        assertNotEquals(cipher1, cipher2)

        val okResult = CloudAccountCrypto.decrypt(cipher1, uid, pin)

        assertEquals(secretPayload, (okResult as CloudDecryptResult.Ok).plain)

        val wrongPinResult = CloudAccountCrypto.decrypt(cipher1, uid, "0000")
        assertTrue(wrongPinResult is CloudDecryptResult.WrongPin)

        val noPinResult = CloudAccountCrypto.decrypt(cipher1, uid, null)
        assertTrue(noPinResult is CloudDecryptResult.PinRequired)
    }

    private val _state = MutableStateFlow(CloudLoginState())
    val state = _state.asStateFlow()

    fun onEmail(value: String) = _state.update { it.copy(email = value, error = null) }
    fun onPass(value: String) = _state.update { it.copy(pass = value, error = null) }
    fun toggleMode() = _state.update { it.copy(isRegisterMode = !it.isRegisterMode, error = null) }
    fun onError(message: String) = _state.update { it.copy(loading = false, error = message) }

    fun signInWithGoogle(idToken: String, onSuccess: () -> Unit) {
        _state.update { it.copy(loading = true, error = null) }
        viewModelScope.launch {
            try {
                container.authManager.signInWithGoogle(idToken)
                _state.update { it.copy(loading = false) }
                onSuccess()
            } catch (e: Exception) {
                _state.update { it.copy(loading = false, error = e.message ?: "فشل تسجيل الدخول بجوجل") }
            }
        }
    }

    fun submit(onSuccess: () -> Unit) {
        val current = _state.value
        if (current.email.isBlank() || current.pass.isBlank()) {
            _state.update { it.copy(error = "يرجى إدخال البريد وكلمة المرور") }
            return
        }
        _state.update { it.copy(loading = true, error = null) }
        viewModelScope.launch {
            try {
                if (current.isRegisterMode) {
                    container.authManager.signUpWithEmail(current.email, current.pass)
                } else {
                    container.authManager.signInWithEmail(current.email, current.pass)
                }
                _state.update { it.copy(loading = false) }
                onSuccess()
            } catch (e: Exception) {
                _state.update { it.copy(loading = false, error = e.message ?: "حدث خطأ") }
            }
        }
    }
}

 * The watchdog samples the video decoder's rendered-frame counter once per second. When the player
 * is READY + playing, a video track is active, no frame has been rendered for [STALL_MS] and the
 * playback position (audio clock) advanced at least [MIN_AUDIO_ADVANCE_MS], it recovers:
 *  1. `stop()` (releases the stuck codec) + `seekToDefaultPosition()` + `prepare()` at the live edge.
 *  2. From the second stall onwards, [onEscalate] may switch the stream container (m3u8 <-> ts).
 * At most [MAX_RECOVERIES] recoveries per [RECOVERY_WINDOW_MS] per item, so still-image / very
 * low-fps channels can never cause a reset loop.
 */
@OptIn(UnstableApi::class)
@Composable
fun LiveVideoStallWatchdog(
    player: ExoPlayer?,
    enabled: Boolean,
    itemKey: Any?,
    onEscalate: () -> Boolean = { false },
) {
    val escalate by rememberUpdatedState(onEscalate)
    LaunchedEffect(player, enabled, itemKey) {
        if (player == null || !enabled) return@LaunchedEffect
        var lastRendered = -1
        var frozenSinceMs = 0L
        var positionAtFreezeMs = 0L
        var recoveries = 0
        var windowStartMs = SystemClock.elapsedRealtime()

        while (isActive) {
            delay(SAMPLE_MS)
            val counters = player.videoDecoderCounters
            val armed = player.isPlaying &&
                player.playbackState == Player.STATE_READY &&
                player.videoFormat != null &&
                counters != null
            if (!armed || counters == null) {
                lastRendered = -1
                continue
            }
            counters.ensureUpdated()
            val rendered = counters.renderedOutputBufferCount
            val now = SystemClock.elapsedRealtime()
            val position = player.currentPosition
            if (rendered != lastRendered) {
                lastRendered = rendered
                frozenSinceMs = now
                positionAtFreezeMs = position
                continue
            }

            val frozenFor = now - frozenSinceMs
            val audioAdvanced = position - positionAtFreezeMs
            if (frozenFor < STALL_MS || audioAdvanced < MIN_AUDIO_ADVANCE_MS) continue

            if (now - windowStartMs > RECOVERY_WINDOW_MS) {
                windowStartMs = now
                recoveries = 0
            }
            if (recoveries >= MAX_RECOVERIES) {
                // Give up for this item: probably a still-image / ultra low-fps channel.
                frozenSinceMs = now
                positionAtFreezeMs = position
                continue
            }
            recoveries++
            Log.w(
                TAG,
                "Live video stall: no frame for ${frozenFor}ms while audio advanced " +
                    "${audioAdvanced}ms (rendered=$rendered, dropped=${counters.droppedBufferCount}) " +
                    "-> recovery #$recoveries",
            )
            // stop() resets the renderers, which releases the deadlocked MediaCodec instance.
            player.stop()
            val escalated = recoveries >= 2 && escalate()
            if (!escalated) {
                player.seekToDefaultPosition()
                player.prepare()
                player.play()
            }
            lastRendered = -1
        }
    }
}

private const val TAG = "SahabLiveWatchdog"
private const val SAMPLE_MS = 1_000L
private const val STALL_MS = 4_500L
private const val MIN_AUDIO_ADVANCE_MS = 3_000L
private const val MAX_RECOVERIES = 3
private const val RECOVERY_WINDOW_MS = 120_000L

/**
 * Shared Media3 (ExoPlayer) factory with:
 * - Singleton 250 MB [SimpleCache] on disk for VOD (Movies & Series episodes) so scrubbing/rewinding/reopening is instant.
 * - Ultra-fast startup buffer thresholds ([DefaultLoadControl]) for sub-second playback start.
 * - Hardware/Software decoder fallback & TS/HLS/DASH/RTSP support.

@OptIn(UnstableApi::class)
object SahabPlayer {

    const val SEEK_STEP_MS = 10_000L
    private const val VIDEO_JOINING_TIME_MS = 3_000L
    private const val MEDIA_CACHE_BYTES = 512L * 1024L * 1024L // 512 MB

    @Volatile
    private var simpleCache: SimpleCache? = null

    @Synchronized
    private fun getMediaCache(context: Context): SimpleCache? {
        simpleCache?.let { return it }
        return runCatching {
            val appContext = context.applicationContext
            val dir = File(appContext.cacheDir, MEDIA_CACHE_DIR).apply { mkdirs() }
            val evictor = LeastRecentlyUsedCacheEvictor(MEDIA_CACHE_BYTES)
            val dbProvider = StandaloneDatabaseProvider(appContext)
            SimpleCache(dir, evictor, dbProvider).also { simpleCache = it }
        }.getOrNull()
    }

    @Synchronized
    fun clearMediaCache(context: Context) {
        runCatching { simpleCache?.release() }
        simpleCache = null
        runCatching {
            File(context.applicationContext.cacheDir, MEDIA_CACHE_DIR).deleteRecursively()
        }
    }

    private const val MEDIA_CACHE_DIR = "sahab_media_cache"
    private val CACHE_KEY_CREDENTIAL_REGEX = Regex("/(live|movie|series)/[^/]+/[^/]+/")

    fun warmUpMediaCache(context: Context) {
        getMediaCache(context)

                .setAllowVideoNonSeamlessAdaptiveness(true)
                .setAllowAudioMixedMimeTypeAdaptiveness(true)
                .setAllowAudioMixedSampleRateAdaptiveness(true)
                .setAllowAudioMixedChannelCountAdaptiveness(true)
                .setExceedVideoConstraintsIfNecessary(true)
                .setExceedRendererCapabilitiesIfNecessary(true)
                .build(),
        )

        val extractors = DefaultExtractorsFactory()
            .setConstantBitrateSeekingEnabled(true)
            .setTsExtractorFlags(
                DefaultTsPayloadReaderFactory.FLAG_ALLOW_NON_IDR_KEYFRAMES or
                    DefaultTsPayloadReaderFactory.FLAG_DETECT_ACCESS_UNITS,
            )

        // (stale .m3u8 + disk write latency). newBuilder() keeps the shared connection pool/dispatcher.

            .setUserAgent("Sahab/1.0 (Android)")

            val cache = getMediaCache(context)
            if (cache != null) {
                val sanitizedKeyFactory = CacheKeyFactory { dataSpec ->
                    val rawKey = dataSpec.key ?: dataSpec.uri.toString()
                    // Isolate cache per account: /kind/<accountHash>/<id>.<ext>
                    // (hash derived from server+user; never stores credentials in cache keys)
                    rawKey.replace(CACHE_KEY_CREDENTIAL_REGEX, "/$1/$accountHash/")

                    .setCache(cache)
                    .setCacheKeyFactory(sanitizedKeyFactory)
                    .setUpstreamDataSourceFactory(httpFactory)
                    .setFlags(CacheDataSource.FLAG_IGNORE_CACHE_ON_ERROR)
            } else {
                httpFactory
            }
        } else {
            httpFactory
        }

        val res = when {
            h >= 2160 -> "4K (${h}p)"
            h >= 1080 -> "FHD (${h}p)"
            h >= 720 -> "HD (${h}p)"
            h > 0 -> "SD (${h}p)"
            else -> format.label ?: "مسار فيديو"
        }
        val fps = if (format.frameRate > 0f) " · ${format.frameRate.roundToInt()}fps" else ""
        val bitrate = if (format.bitrate > 0) " · ${String.format(Locale.US, "%.1f", format.bitrate / 1_000_000f)} Mbps" else ""
        return "$res$fps$bitrate"
    }

    fun activeBitrateText(tracks: Tracks, isLive: Boolean): String {
        val format = tracks.groups
            .firstOrNull { it.type == C.TRACK_TYPE_VIDEO && it.isSelected }
            ?.let { g -> (0 until g.length).firstOrNull { g.isTrackSelected(it) }?.let { g.getTrackFormat(it) } }
        if (format != null && format.bitrate > 0) {
            return String.format(Locale.US, "%.1f Mbps", format.bitrate / 1_000_000f)
        }
        return ""
    }

    fun activeAudioLabel(tracks: Tracks): String {
        val format = tracks.groups
            .firstOrNull { it.type == C.TRACK_TYPE_AUDIO && it.isSelected }
            ?.let { g ->
                (0 until g.length).firstOrNull { g.isTrackSelected(it) }?.let { g.getTrackFormat(it) }
                    ?: if (g.length > 0) g.getTrackFormat(0) else null
            } ?: return "الصوت الافتراضي"

        val langCode = format.language
        val langName = if (!langCode.isNullOrBlank() && langCode != "und") {
            Locale(langCode).getDisplayLanguage(Locale("ar")).ifBlank { langCode }
        } else {
            format.label?.takeIf { it.isNotBlank() } ?: "الصوت"
        }
        val codec = codecShortName(format)
        return if (codec != null) "$langName ($codec)" else langName
    }

    fun activeSubtitleLabel(tracks: Tracks, disabled: Boolean): String {
        if (disabled) return "بدون ترجمة"
        val format = tracks.groups
            .firstOrNull { it.type == C.TRACK_TYPE_TEXT && it.isSelected }
            ?.let { g -> (0 until g.length).firstOrNull { g.isTrackSelected(it) }?.let { g.getTrackFormat(it) } }
            ?: return "بدون ترجمة"
        val langCode = format.language
        val langName = if (!langCode.isNullOrBlank() && langCode != "und") {
            Locale(langCode).getDisplayLanguage(Locale("ar")).ifBlank { langCode }
        } else {
            format.label?.takeIf { it.isNotBlank() } ?: "ترجمة"
        }
        return "$langName (CC)"
    }

    fun activeQualityButtonLabel(tracks: Tracks, hasOverride: Boolean, isLive: Boolean): String {
        val format = tracks.groups
            .firstOrNull { it.type == C.TRACK_TYPE_VIDEO && it.isSelected }
            ?.let { g -> (0 until g.length).firstOrNull { g.isTrackSelected(it) }?.let { g.getTrackFormat(it) } }
        val h = format?.height ?: 0
        val res = when {
            h >= 2160 -> "4K"
            h >= 1080 -> "1080p FHD"
            h >= 720 -> "720p HD"
            h > 0 -> "${h}p"
            else -> ""
        }
        return when {
            res.isBlank() -> if (hasOverride) "الجودة" else "تلقائي"
            hasOverride -> res
            else -> "تلقائي $res"
        }
    }

    private fun codecShortName(format: Format): String? {
        val mime = format.sampleMimeType ?: return null
        return when (mime) {
            MimeTypes.AUDIO_AAC -> "AAC"
            MimeTypes.AUDIO_AC3 -> "AC3"
            MimeTypes.AUDIO_E_AC3, MimeTypes.AUDIO_E_AC3_JOC -> "EAC3"
            MimeTypes.AUDIO_MPEG -> "MP3"
            MimeTypes.AUDIO_OPUS -> "OPUS"
            MimeTypes.AUDIO_VORBIS -> "Vorbis"
            MimeTypes.AUDIO_FLAC -> "FLAC"
            else -> mime.substringAfter('/', "").uppercase(Locale.US).takeIf { it.isNotBlank() }
        }
    }
